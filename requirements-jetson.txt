# Jetson Orin Nano / NX overrides — JetPack 6.x (L4T r36.3 / 36.4), Python 3.10.
#
# JetPack 6 ships Python 3.10.12 as the system interpreter and JetPack's
# pre-built OpenCV bindings are linked against it, so we deliberately use 3.10
# (not 3.11 from deadsnakes) — that keeps GStreamer + NVMM-enabled OpenCV
# available "for free" via --system-site-packages.
#
# Sources of Jetson-specific wheels (in order of preference):
#   1. https://pypi.jetson-ai-lab.dev/      ← maintained pip index, what we use
#   2. https://elinux.org/Jetson_Zoo         ← community wiki, lags behind for JP 6.1+
#   3. Build from source                     ← last resort, hours on Orin Nano
#
# The Jetson AI Lab index segments wheels by (JetPack-major × CUDA-version):
#     /jp6/cu122/   →  JetPack 6.0           (CUDA 12.2)
#     /jp6/cu126/   →  JetPack 6.1 and 6.2   (CUDA 12.6)
#     /jp5/cu114/   →  JetPack 5.x           (legacy)
#
# Confirm the CUDA version on your device with `nvidia-smi` before installing.
#
# ── Usage ──────────────────────────────────────────────────────────────────
#
#   # 1. Create the venv WITH system-site-packages so JetPack's GStreamer-
#   #    enabled OpenCV is visible inside it. Use the system python3.10.
#   python3.10 -m venv --system-site-packages .venv
#   source .venv/bin/activate
#   pip install --upgrade pip wheel
#
#   # 2. Install the project + extras (do NOT include [ai] — its CPU
#   #    onnxruntime + headless OpenCV wheels are the things we are overriding).
#   pip install -e ".[sensors,dashboard]"
#   pip install scikit-learn>=1.5    # the only [ai] dep we still want from PyPI
#
#   # 3. Drop anything that may have leaked in from a prior install attempt.
#   pip uninstall -y onnxruntime opencv-python opencv-python-headless ultralytics || true
#
#   # 4. Apply the Jetson-specific pins. The --extra-index-url tells pip to
#   #    look at the Jetson AI Lab index in ADDITION to PyPI — pip picks the
#   #    Jetson aarch64 wheel for onnxruntime-gpu and falls back to PyPI for
#   #    everything else.
#   pip install -r requirements-jetson.txt
#
#   # 5. Smoke test — both should print without error.
#   python -c "import cv2; print('cv2', cv2.__version__, '| GStreamer:', 'GSTREAMER' in cv2.getBuildInformation())"
#   python -c "import onnxruntime as ort; print(ort.get_available_providers())"
#   # Expect:
#   #   cv2 4.8.0 | GStreamer: True
#   #   ['TensorrtExecutionProvider', 'CUDAExecutionProvider', 'CPUExecutionProvider']
#
# ── Notes ──────────────────────────────────────────────────────────────────
# - `ultralytics` is intentionally NOT reinstalled. It's only needed for
#   training and the one-shot YOLOv8n→ONNX export — do those on a dev box.
# - If you later upgrade JetPack across CUDA majors (e.g. 6.0 → 6.1 switches
#   CUDA 12.2 → 12.6), change the --extra-index-url path and re-install the
#   wheel — CUDA ABI changes between minor releases would otherwise surface
#   as a silent CPU fallback or an import-time load failure.

# ── Extra index for Jetson-built wheels ────────────────────────────────────
# Change cu126 → cu122 if you are on JetPack 6.0.
--extra-index-url https://pypi.jetson-ai-lab.dev/jp6/cu126

# ── ONNX Runtime — Jetson GPU build (CUDA + TensorRT execution providers) ──
# Leave the version unpinned: the Jetson AI Lab index serves the latest cp310
# wheel for the (jp6, cu126) channel, currently ORT 1.20.x. If you want to
# pin, check what `pip index versions onnxruntime-gpu` returns first.
onnxruntime-gpu

# ── Everything else the runtime needs that wasn't already pulled in ────────
# Pinned loosely so a future `pip install -U` doesn't silently re-introduce
# CPU onnxruntime as a transitive dep.
numpy>=1.26,<2.0          # Jetson PyPI builds of numpy 2.x can be flaky; stay on 1.x
scikit-learn>=1.5          # DBSCAN for huddling detector
