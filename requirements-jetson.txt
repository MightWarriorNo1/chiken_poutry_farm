# Jetson Orin Nano / NX overrides — JetPack 6.x (L4T r36.x), Python 3.10.
#
# JetPack 6 ships Python 3.10.12 + an OpenCV with GStreamer enabled (via
# `nvidia-opencv`). We deliberately use the system python (not 3.11) so the
# venv inherits that OpenCV through `--system-site-packages`.
#
# ── Inference backends ──────────────────────────────────────────────────────
#
# Two paths supported by the inference factory (auto-dispatched by artifact
# extension):
#
#   • `.pt` checkpoint → `UltralyticsBirdDetector` → torch.cuda on the iGPU
#                        (the production path on this Jetson).
#   • `.onnx` artifact → `YoloBirdDetector` → ONNX Runtime, with
#                        TensorRT/CUDA EPs when an ORT-GPU wheel is installed.
#
# For the PoC we pin to the `.pt` / ultralytics path. ORT here stays CPU; it
# is only used if someone drops an `.onnx` into models/.
#
# ── One-time GPU setup (Jetson PyTorch + ultralytics) ──────────────────────
#
# NVIDIA hosts Jetson-built torch + torchvision wheels at
#   https://developer.download.nvidia.com/compute/redist/jp/v62/pytorch/
# These rotate; the exact filenames change between l4t patch releases. Open
# that directory in a browser, copy the URLs that match `cp310` + your L4T,
# then on the Jetson (with the venv active):
#
#   pip install <torch-2.x.x-cp310-cp310-linux_aarch64.whl URL>
#   pip install <torchvision-0.x.x-cp310-cp310-linux_aarch64.whl URL>
#
# Verify CUDA is wired in before installing ultralytics — otherwise pip will
# happily pull a PyPI torch wheel that has no CUDA:
#
#   python -c "import torch; print(torch.__version__, 'cuda?', torch.cuda.is_available())"
#   # Expect: 2.x.x  cuda? True
#
# Then:
#
#   pip install -r requirements-jetson.txt    # picks up ultralytics + wsdiscovery
#
# Verify the YOLO path is hot:
#
#   python -c "from ultralytics import YOLO; import torch; \
#              m=YOLO('models/bird-detector/v1.0.0/model.pt'); \
#              print('device:', m.device, 'cuda:', torch.cuda.is_available())"
#
# ── Usage ──────────────────────────────────────────────────────────────────
#
#   python3.10 -m venv --system-site-packages .venv
#   source .venv/bin/activate
#   pip install --upgrade pip wheel
#   pip install -e ".[sensors,dashboard]"
#   pip install "scikit-learn>=1.5"
#   pip uninstall -y onnxruntime opencv-python opencv-python-headless || true
#   # ── Install Jetson torch + torchvision FIRST (see comment above) ──
#   pip install -r requirements-jetson.txt
#
# Smoke test:
#   python -c "import cv2; print(cv2.__version__, 'GStreamer:', \
#              'GStreamer:                   YES' in cv2.getBuildInformation())"
#   python -c "import torch; print(torch.cuda.is_available())"
#   python -c "from ultralytics import YOLO; print('ok')"

# ── CPU ONNX Runtime (fallback / .onnx artifacts) ──────────────────────────
onnxruntime>=1.18

# ── Ultralytics YOLO — drives the .pt → torch.cuda path ────────────────────
# Pin loosely; ultralytics tracks its own torch compatibility matrix.
ultralytics>=8.3,<9

# ── ONVIF WS-Discovery for the RTSP camera auto-discovery flow ─────────────
# Pure-Python, small dep — used by `src/edge/dashboard/discovery.py`.
WSDiscovery>=2.0

# ── Misc pins (stop pip from re-introducing CPU onnxruntime as a transitive) ──
numpy>=1.26,<2.0
scikit-learn>=1.5
